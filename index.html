app/ml.py — keyword explainer
def _keywords(self, row, class_idx: int, top_n: int) -> list[dict]:
    """Terms in the input that pushed the prediction towards the class."""
    if self._weights is None or row.nnz == 0:
        return []
    contrib = row.data * self._weights[class_idx, row.indices]
    order = np.argsort(contrib)[::-1][:top_n]
    return [{"term": str(self.vocabulary[row.indices[i]]),
             "weight": round(float(contrib[i]), 4)}
            for i in order if contrib[i] > 0]

import { Content, GoogleGenerativeAI } from "@google/generative-ai"

const GENAI_MODEL_NAME = 'gemini-2.0-flash';

export function generateTextResponse(historyConversation: Content[]) {
  return new Promise(async (resolve, reject) => {
    try {
      const genAI = new GoogleGenerativeAI(GENAI_API_KEY);
      const model = genAI.getGenerativeModel({
        model: GENAI_MODEL_NAME,
        systemInstruction: { role: "system", parts: [...] }
      });
      const result = await model.generateContent({
        contents: historyConversation
      })
      return resolve(result.response.text());
    } catch (err) {
      return reject(GENAI_ERROR);
    }
  })
}
